Consulta SQL do Banco de Dados

SELECT s.aluno_id, s.nome, c.nome_curso, e.nota
FROM Alunos s
JOIN Matriculas e ON s.aluno_id = e.aluno_id
JOIN Cursos c ON e.curso_id = c.curso_id
WHERE e.nota >= 7.0
ORDER BY e.nota DESC;

Algoritmo Bubble Sort em JavaScript

function bubbleSort(arr) {
    let len = arr.length;
    for (let i = 0; i < len; i++) {
        for (let j = 0; j < len - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                let temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
    return arr;
}

Salvamento no Local Storage

const salvarTarefas = (tarefas) => {
    try {
        const jsonTarefas = JSON.stringify(tarefas);
        localStorage.setItem('minhas_tarefas', jsonTarefas);
    } catch (error) {
        console.error("Erro ao salvar no storage:", error);
    }
};
